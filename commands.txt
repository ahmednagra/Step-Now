# =============================================================================
#  step-now.de on Laragon (Windows) — full local setup
# =============================================================================

# -----------------------------------------------------------------------------
# 1. PUT THE PROJECT IN THE LARAGON WWW FOLDER
# -----------------------------------------------------------------------------
# Laragon auto-creates a *.test domain for every folder under D:\laragon\www\
# So the folder name MATTERS — name it "step-now" and you get http://step-now.test

cd D:\laragon\www
git clone <repo-url> step-now
cd step-now


# -----------------------------------------------------------------------------
# 2. CREATE THE DATABASE (this is what fixes the Connection.php:825 error)
# -----------------------------------------------------------------------------
# Open Laragon → click "Database" button → it opens HeidiSQL
# In HeidiSQL run:

CREATE DATABASE stepnow_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

# OR from Laragon Terminal:
mysql -u root -e "CREATE DATABASE IF NOT EXISTS stepnow_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;"


# -----------------------------------------------------------------------------
# 3. .env CONFIG FOR LARAGON
# -----------------------------------------------------------------------------
# Open .env and set these values (key changes for Laragon shown below):

APP_URL=http://step-now.test          # NOT localhost — Laragon serves it here
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=stepnow_db
DB_USERNAME=root
DB_PASSWORD=                          # Laragon's MySQL has empty root password

# Mailpit — way better than MAIL_MAILER=log for testing the DOI flow
MAIL_MAILER=smtp
MAIL_HOST=127.0.0.1
MAIL_PORT=1025
MAIL_USERNAME=null
MAIL_PASSWORD=null
MAIL_ENCRYPTION=null
MAIL_FROM_ADDRESS="info@step-now.de"
MAIL_FROM_NAME="StepNow Rides"


# -----------------------------------------------------------------------------
# 4. INSTALL DEPENDENCIES (Laragon Terminal — opens in project folder)
# -----------------------------------------------------------------------------

composer install
npm install
npm run build

php artisan key:generate
php artisan storage:link


# -----------------------------------------------------------------------------
# 5. RUN MIGRATIONS + SEED
# -----------------------------------------------------------------------------

php artisan migrate
php artisan db:seed


# -----------------------------------------------------------------------------
# 6. RELOAD LARAGON SO IT REGISTERS step-now.test
# -----------------------------------------------------------------------------
# In the Laragon window: click "Reload" (top right, next to port 80)
# OR right-click Laragon system tray → Apache → Reload


# -----------------------------------------------------------------------------
# 7. OPEN THE SITE
# -----------------------------------------------------------------------------
# Front:    http://step-now.test
# Login:    http://step-now.test/login
# Mailpit:  http://localhost:8025      (see DOI confirmation mails arrive here)


# -----------------------------------------------------------------------------
# 8. AFTER PULLING NEW PATCHES (PolicySeeder etc. updates)
# -----------------------------------------------------------------------------

php artisan migrate
php artisan db:seed --class=Database\\Seeders\\PolicySeeder --force
php artisan db:seed --class=Database\\Seeders\\SettingsTableSeeder --force
php artisan view:clear
php artisan route:clear
php artisan config:clear


# -----------------------------------------------------------------------------
# 9. DESTRUCTIVE FULL RESET (local only, drops everything)
# -----------------------------------------------------------------------------

php artisan migrate:fresh --seed


# -----------------------------------------------------------------------------
# 10. TEST THE NEWSLETTER DOUBLE-OPT-IN FLOW
# -----------------------------------------------------------------------------
# 1. Open http://step-now.test
# 2. Sign up to the newsletter with any email + tick the consent box
# 3. Open http://localhost:8025  (Mailpit web UI)
# 4. Click the confirmation link in the mail
# 5. Verify the row in DB:

mysql -u root -e "USE stepnow_db; SELECT email, confirmed_at FROM newsletters;"


# -----------------------------------------------------------------------------
# 11. IF YOU STILL HIT THE Connection.php:825 ERROR
# -----------------------------------------------------------------------------
# Run these checks in the Laragon Terminal:

# (a) Is MySQL actually up?
mysql -u root -e "SELECT VERSION();"
# Expected: 8.4.3

# (b) Does the database exist?
mysql -u root -e "SHOW DATABASES LIKE 'stepnow_db';"
# Expected: one row

# (c) Does Laravel see the .env?
php artisan config:clear
php artisan tinker
>>> config('database.connections.mysql')
>>> exit
# The DB host/name printed must match what you set in .env

# (d) MySQL 8.4 auth issue — if Laravel says "auth plugin caching_sha2_password":
mysql -u root -e "ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY '';"
mysql -u root -e "FLUSH PRIVILEGES;"
